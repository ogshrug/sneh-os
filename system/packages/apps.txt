# Optional apps, chosen when flashing (APPS=files,vlc in sneh.conf) and
# changeable later with `sne apps add|remove NAME`.

vlc:     vlc vlc-plugins-base vlc-plugin-ffmpeg   # VLC media player (~150 MB disk with Qt)
mpv:     mpv                                # mpv: a minimal, much lighter video player

images:  imv                                # imv: a tiny image viewer
